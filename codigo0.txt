===== ./blueprints/books/models.py =====
class Livro:

    def __init__(self, nome_livro, autor_livro, valor_livro, categoria_livro,livro_id=None):
        self.__nome = nome_livro
        self.__autor = autor_livro
        self.__valor = float(valor_livro)
        self.__categoria = categoria_livro
        self.__id = livro_id

    
    def to_dict(self):
        return {
            "nome": self.nome,
            "autor": self.autor,
            "valor": self.valor,
            "categoria": self.categoria,
            "id":self.id
        }
    
    #getter utilizado para recuperar valor 
    @property
    def nome(self):
        return str.capitalize(self.__nome)
    
    #utilizado para atribuir valor
    @nome.setter
    def nome(self, valor):
        self.__nome = str.capitalize(valor)

    @property
    def autor(self):
        return self.__autor
    
    @autor.setter
    def autor(self, valor):
        self.__autor = str.upper(valor)

    @property
    def categoria(self):
        return self.__categoria
    
    @categoria.setter
    def categoria(self, valor):
        self.__categoria = valor

    @property
    def valor(self):
        return self.__valor
    
    @valor.setter
    def valor(self, v):
        self.__valor = v

    @property
    def id(self):
        return self.__id
    
    @id.setter
    def id(self, v):
        self.__id = v

===== ./blueprints/books/dao.py =====
import json
import os
from .models import Livro

class LivroDAO:

    #construtor
    def __init__(self, ac):
        self.__arquivo_caminho = ac

    #getter
    @property
    def arquivo_caminho(self):
        return self.__arquivo_caminho
    
    #setter
    @arquivo_caminho.setter
    def arquivo_caminho(self, v):
        self.__arquivo_caminho = v

    def carregar_livros(self):
        if not os.path.exists(self.__arquivo_caminho):
            return []

        try:
            with open(self.__arquivo_caminho, "r", encoding="utf-8") as f:
                dados_json = json.load(f)
                lista = []

                for item in dados_json:
                    nome = item.get("nome")
                    autor = item.get("autor")
                    valor = item.get("valor")
                    categoria = item.get("categoria")
                    id = item.get("id")

                    livro = Livro(nome,autor,valor,categoria,livro_id=id)
                    lista.append(livro)
                return lista
        except (json.JSONDecodeError, FileNotFoundError):
            return []


    def salvar_livro(self, novo_livro):
        """Adiciona um novo livro ao arquivo JSON mantendo os antigos."""
        livros = self.carregar_livros()

        if len(livros) > 0:            
            livro_id = livros[-1].id + 1
        else:
            livro_id = 1

        novo_livro.id = livro_id

        livros.append(novo_livro)

        lista_json = [l.to_dict() for l in livros]

        with open(self.__arquivo_caminho, "w", encoding="utf-8") as f:
            json.dump(lista_json, f, indent=4, ensure_ascii=False)
===== ./blueprints/books/controllers.py =====
from flask import Flask, render_template, request, redirect, url_for, flash

from .dao import LivroDAO

from .models import Livro

class LivroController:

    def __init__(self):
        self.__dao = LivroDAO("livros.json")


    def listar_livros(self):
        lista = self.__dao.carregar_livros()
        return render_template("livros.html", livros=lista)
    
    def cadastrar_livro(self):
        # Pegando os dados do formulário
        nome = request.form.get("nome")
        autor = request.form.get("autor")
        valor = request.form.get("valor")
        categoria = request.form.get("categoria")

        # Validações Simples
        if not nome or not autor or not valor or not categoria:
            flash("Erro: Todos os campos são obrigatórios!", "danger")
        else:
            try:
                valor_float = float(valor)
                if valor_float <= 0:
                    flash("Erro: O valor deve ser maior que zero!", "danger")
                else:
                    # Se tudo estiver certo, salvamos
                    novo_livro = Livro(nome,autor,valor,categoria)


                    self.__dao.salvar_livro(novo_livro)
                    flash(f"Sucesso: O livro '{nome}' foi cadastrado!", "success")
                    return redirect(url_for("books.listar_livros"))
            except ValueError:
                flash("Erro: Digite um número válido para o valor.", "danger")

        return self.preparar_cadastro()
    
    def preparar_cadastro(self):
        return render_template("cadastrar.html")
